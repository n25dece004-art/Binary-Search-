bool binarySearch(int a[], int n, int x) { int l = 0, r = n - 1; while (l <= r) { int m = (l + r) / 2; if (a[m] == x) { return true; } else if (a[m] < x) { l = m + 1; // Tìm nửa phải } else { r = m - 1; // Tìm nửa trái } } return false; }
